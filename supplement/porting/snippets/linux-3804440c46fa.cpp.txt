Result<Count> doPwrite(File f, Bytes b, std::uint64_t offset) {
    if (offset > static_cast<std::uint64_t>(std::numeric_limits<off_t>::max())) return posixFailure<Count>(EOVERFLOW);
    const auto r = ::pwrite(f.native, b.data(), std::min(b.size(), static_cast<std::size_t>(PTRDIFF_MAX)), static_cast<off_t>(offset));
    if (r < 0) return posixFailure<Count>();
    return ok(static_cast<Count>(r));
}
