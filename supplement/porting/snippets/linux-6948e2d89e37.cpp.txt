Result<Count> doSend(Socket s, Bytes b, WriteMode mode) {
    const int flags = mode == WriteMode::urgent ? MSG_OOB : 0;
    const auto r = ::send(s.native, b.data(), std::min(b.size(), static_cast<std::size_t>(PTRDIFF_MAX)), flags);
    if (r < 0) return posixFailure<Count>();
    return ok(static_cast<Count>(r));
}
