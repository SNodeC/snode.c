namespace core::system {
enum class ErrorDomain { none, posix, win32, winsock, resolver, ntstatus, loader };
struct Error { ErrorDomain domain{}; std::uint32_t code{}; std::string text; };
template<class T> struct Result {
    T value{};
    Error error{};
    explicit operator bool() const { return error.domain == ErrorDomain::none; }
};
struct Unit {};
using Status = Result<Unit>;
template<class T> Result<T> ok(T value) { return {std::move(value), {}}; }
inline Status ok() { return {}; }
template<class T = Unit> Result<T> fail(ErrorDomain d, std::uint32_t e) {
    return {{}, {d, e}};
}
template<class T = Unit> Result<T> posixFailure(int e = errno) {
    return fail<T>(ErrorDomain::posix, static_cast<std::uint32_t>(e));
}
inline Status posixStatus(int r) { return r == -1 ? posixFailure() : ok(); }
using Count = std::ptrdiff_t;
using Bytes = std::span<const std::byte>;

#ifdef _WIN32
template<class T = Unit> Result<T> winFailure(DWORD e = GetLastError()) {
    return fail<T>(ErrorDomain::win32, e);
}
template<class T = Unit> Result<T> wsaFailure(int e = WSAGetLastError()) {
    return fail<T>(ErrorDomain::winsock, static_cast<std::uint32_t>(e));
}
inline Status winStatus(BOOL r) { return r ? ok() : winFailure(); }
inline Status socketStatus(int r) { return r == SOCKET_ERROR ? wsaFailure() : ok(); }
#else
inline Status socketStatus(int r) { return posixStatus(r); }
#endif
